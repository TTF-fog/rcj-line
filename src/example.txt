// Four-motor test for two Cytron Maker Drive boards
// No external libraries required.
#include "Arduino.h"

struct Motor {
    int pin1;
    int pin2;
};

Motor motors[] = {
    {23, 22},    // Motor 1
    {12, 11},  // Motor 2
    {9, 8},    // Motor 3
    {7, 6}     // Motor 4
};

const int MOTOR_COUNT = 4;
const int SPEED = 100;          // PWM: 0-255
const int RUN_TIME = 1500;      // milliseconds
const int PAUSE_TIME = 1000;

void stopMotor(int i) {
    analogWrite(motors[i].pin1, 0);
    analogWrite(motors[i].pin2, 0);
}

void driveMotor(int i, int speed) {
    // Positive = one direction; negative = opposite
    if (speed > 0) {
        analogWrite(motors[i].pin1, speed);
        analogWrite(motors[i].pin2, 0);
    } else if (speed < 0) {
        analogWrite(motors[i].pin1, 0);
        analogWrite(motors[i].pin2, -speed);
    } else {
        stopMotor(i);
    }
}

void setup() {
    Serial.begin(115200);

    for (int i = 0; i < MOTOR_COUNT; i++) {
        pinMode(motors[i].pin1, OUTPUT);
        pinMode(motors[i].pin2, OUTPUT);
        stopMotor(i);
    }

    Serial.println("Four-motor test starting");
    delay(1000);
}

void loop() {
    for (int i = 0; i < MOTOR_COUNT; i++) {
        Serial.print("Motor ");
        Serial.print(i + 1);
        Serial.println(" forward");

        driveMotor(i, SPEED);
        delay(RUN_TIME);
        stopMotor(i);
        delay(PAUSE_TIME);

        Serial.print("Motor ");
        Serial.print(i + 1);
        Serial.println(" reverse");

        driveMotor(i, -SPEED);
        delay(RUN_TIME);
        stopMotor(i);
        delay(PAUSE_TIME);
    }

    Serial.println("Test complete. Repeating in 3 seconds.");
    delay(3000);
}
